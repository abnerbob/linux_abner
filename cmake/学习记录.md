一、 配置和编译
```
配置  cmake -S source_dir -B build_dir
编译 cmake --build build_dir
install cmake --install build_dir (3.15+) / cmake --build build_dir --target install
编译target  cmake --build build_dir --target target_name

配置省略目录
cmake . 当前目录为源码目录和编译目录
cmake source_dir 当前目录为build目录( mkdir build; cd build; cmake ..)

编译用make
make -C build_dir
make -C build install

单命令环境变量赋值 CC=clang CXX=clang++ cmake ..
cmake -D指定编译器 cmake -DCMAKE_CXX_COMPILER=clang++ .. 

-D传入cache变量
-D <var>:<type>=<value>, -D <var>=<value>

指定编译器 cmake -G "Ninja" .. (默认生成unix makefile，可以指定生成ninja工程文件或者其他的)
指定生成器 cmake -G "Visual Studio 17 2022" -T "ClangCL" .. (常见windows场景，-T参数)

-L[A][H] 列出非高级缓存变量 
如果指定了``A``，那么它也会显示高级变量。如果指定了``H``，它还会显示每个变量的帮助。
Advanced Cache变量默认隐藏。mark_as_advanced设置其属性

-E 运行内置命令行
cmake脚本中常用 ${CMAKE_COMMAND} -E
cmake -E help 查看帮助

-P 运行脚本
直接运行脚本，没有执行配置或生成步骤，也没有修改缓存。如果使用“-D”定义变量，则必须在“-P”参数之前完成。
```

三、名称空间
本地场景 别名目标
my_project/
├── CMakeLists.txt         (顶层)
├── math/
│   ├── CMakeLists.txt     (库层)
│   ├── math.c
│   └── include/math.h
└── app/
    ├── CMakeLists.txt     (应用层)
    └── main.c

```
# 1. 定义真实的底层目标（通常以工程名或模块名加后缀，防止冲突）
add_library(math_core STATIC math.c)

# 2. 设置包含目录（极其重要：把头文件路径绑定到目标上）
target_include_directories(math_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

# 3. 施展魔法：创建一个带有名称空间的别名（ALIAS）！
# 格式：add_library(<Namespace>::<TargetName> ALIAS <RealTarget>)
add_library(MyOrg::math ALIAS math_core)
```

```
add_executable(my_app main.c)

# 直接链接名称空间目标！
target_link_libraries(my_app PRIVATE MyOrg::math)
```

在 CMake 中，一旦一个目标名字里带有 ::，CMake 就会将其视为 “导入目标（Imported Target）” 或 “别名目标（Alias Target）”。
这意味着：任何人都不允许在外部修改它的属性。

外部场景 install export场景
生产者
```
cmake_minimum_required(VERSION 3.10)
project(MathLib VERSION 1.0)

add_library(math_obj STATIC src/math.c)

# 1. 核心：定义目标的包含属性。
# BUILD_INTERFACE: 内部开发用路径；INSTALL_INTERFACE: 安装后的相对路径。
target_include_directories(math_obj PUBLIC 
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include> 
)

# 2. 将目标分配到一个“导出名字”中（这里叫 MathLibTargets）
install(TARGETS math_obj
        EXPORT MathLibTargets  # 就像把书放进特定的书架分类
        LIBRARY DESTINATION lib
        ARCHIVE DESTINATION lib
        RUNTIME DESTINATION bin
        INCLUDES DESTINATION include
)

# 3. 关键：生成并安装导出的说明书，并在这里指定“名称空间”！
# 这里的 NAMESPACE 就是别人以后看到的双冒号前缀
install(EXPORT MathLibTargets
        FILE MathLibTargets.cmake
        NAMESPACE MyOrg::        
        DESTINATION lib/cmake/MathLib
)

# 4. 生成一个简单的 Config 文件，让 find_package 能工作
include(CMakePackageConfigHelpers)
write_basic_config_version_file("MathLibConfigVersion.cmake" VERSION 1.0 COMPATIBILITY AnyNewerVersion)
install(FILES "MathLibConfigVersion.cmake" DESTINATION lib/cmake/MathLib)

# 这里的 Config.cmake 内容通常很简单，就是包含上面的导出说明书
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/MathLibConfig.cmake" "include(\"\${CMAKE_CURRENT_LIST_DIR}/MathLibTargets.cmake\")")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/MathLibConfig.cmake" DESTINATION lib/cmake/MathLib)
```
消费者
```
cmake_minimum_required(VERSION 3.10)
project(MyApp)

# 1. 寻找已安装的 MathLib
# CMake 会去系统的 CMAKE_PREFIX_PATH 下找 MathLibConfig.cmake
find_package(MathLib REQUIRED)

add_executable(my_app main.c)

# 2. 直接使用名称空间目标！
# 虽然生产者代码里目标叫 math_obj，但因为 NAMESPACE MyOrg:: 的存在，
# 消费者这里必须且只能用 MyOrg::math_obj
target_link_libraries(my_app PRIVATE MyOrg::math_obj)
```

六、cmake编译ko的函数封装
```
# KernelHelpers.cmake

function(add_kernel_module TARGET_NAME)
    # 1. 定义解析参数
    cmake_parse_arguments(KO "" "KDIR" "SRCS;DEFINES;INCLUDES" ${ARGN})

    # 2. 转换 CMake 的 INCLUDES 和 DEFINES 为 Kbuild 格式
    set(KBUILD_CFLAGS "")
    foreach(inc ${KO_INCLUDES})
        list(APPEND KBUILD_CFLAGS "-I${inc}")
    endforeach()
    foreach(def ${KO_DEFINES})
        list(APPEND KBUILD_CFLAGS "-D${def}")
    endforeach()

    # 3. 生成 Kbuild 文件 (核心：将 CFLAGS 注入 ccflags-y)
    set(KBUILD_FILE "${CMAKE_CURRENT_BINARY_DIR}/Kbuild")
    file(WRITE ${KBUILD_FILE} "obj-m := ${TARGET_NAME}.o\n")
    file(APPEND ${KBUILD_FILE} "ccflags-y := ${KBUILD_CFLAGS}\n")

    # 假设所有源文件都要编译进这个 KO
    foreach(src ${KO_SRCS})
        get_filename_component(src_we ${src} NAME_WE)
        file(APPEND ${KBUILD_FILE} "${TARGET_NAME}-y += ${src_we}.o\n")
    endforeach()

    # 4. 创建构建目标
    add_custom_target(${TARGET_NAME} ALL
        COMMAND $(MAKE) -C ${KO_KDIR} M=${CMAKE_CURRENT_BINARY_DIR} src=${CMAKE_CURRENT_SOURCE_DIR} modules
        WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
        BYPRODUCTS "${TARGET_NAME}.ko"
        VERBATIM
    )
endfunction()

使用
include(KernelHelpers.cmake)

# 定义内核模块
add_kernel_module(my_driver
    KDIR "/lib/modules/${KERNEL_VERSION}/build"
    SRCS main.c hardware.c
    INCLUDES 
        ${CMAKE_CURRENT_SOURCE_DIR}/include
        ${PROJECT_SOURCE_DIR}/common/include
    DEFINES
        DEBUG_LOG
        CHIP_VERSION=2
)
```

七、cmake编排
ExternalProject_Add模块
include(ExternalProject)
1. 原生cmake模式
```
ExternalProject_Add(ext_cmake_lib
    SOURCE_DIR ${SOURCE_DIR}
    # 核心：透传主工程的工具链和变量
    CMAKE_ARGS 
        -DCMAKE_C_COMPILER=${CMAKE_C_COMPILER}
        -DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE}
        -DEXPORT_SYMBOLS=ON
    BUILD_ALWAYS TRUE
)
```
2. 自定义命令模式
```
ExternalProject_Add(ext_legacy_lib
        SOURCE_DIR ${SOURCE_DIR}
# 核心：完全手动控制，CMake 只负责“踢一脚”
CONFIGURE_COMMAND ./configure --host=arm-linux --prefix=<INSTALL_DIR>
BUILD_COMMAND $(MAKE)
INSTALL_COMMAND $(MAKE) install )
BUILD_ALWAYS TRUE
)

### 为什么这两种形式能统合在一起？
`ExternalProject_Add` 的巧妙之处在于它的**默认行为**：
> 如果你不写 `CONFIGURE_COMMAND`，它默认就认为是路线 1（调 `cmake`）。
> 如果你写了 `CONFIGURE_COMMAND`，它就自动切换到路线 2（调你的命令）。
使用了CONFIGURE_COMMAND，表示自定义命令模式，且每个阶段都有标志文件生成。
```

自定义命令模式，编译调试阶段失败，成功的阶段都会跳过，继续调试，但是二次编译场景即使更新了文件也不会触发编译。
可以设置BUILD_ALWAYS TRUE强制触发build_command及以后命令。
可以传入$(MAKE)，使用make的jobserver机制，处理顶层和外部模块的并行统一调度。

原生cmake的场景，二次构建场景，会默认执行cmake --build build_dir，能识别到外部模块更新。
原生cmake由顶层统一调度并发。

ExternalProject_Add调用原生cmake外部模块时候，可以USES_TERMINAL_BUILD设置成1(将外部项目的构建过程直接挂载到当前终端的控制权下 不理解先记录)
调用原生cmake外部模块，可以使用Jobserver 机制(共享并发令牌)
ExternalProject_Add调原生和非原生的cmake工程，其实本质上还是用了子工程的增量编译来感知更新。

3. add_subdirectory (或 FetchContent) 模式
接添加原生cmake工程子目录的方式，target直接解析到上层cmake中，编译由上层cmake统一调度。
工具链和配置项使用上层cmake工程。
不适用异构(不同工具链，不同平台)组件。

4. add_custom_target
原生接口，执行指定的target，可以用来执行外部组件的编译和特定任务
调试过程中，target中的每个任务都会顺序执行，复杂执行流程会很尴尬。 
无法感知组件内部的更新，但每次执行make target都会执行。

八、include_guard
include_guard(GLOBAL)：
作用：在整个构建树（Global Build）中只执行一次。
场景：适用于全局单例的配置，比如定义全局工具函数。
include_guard(DIRECTORY)：
作用：在当前目录及其子目录的作用域内只执行一次。
include_guard() (不带参数)：
作用：默认行为。它取决于当前的变量作用域。

九、配置流程耗时分析和优化
1. --profiling-output 参数打印perf 分析耗时操作
cmake .. --profiling-output=perf.json --profiling-format=google-trace
运行配置时开启追踪模式
cmake .. --trace-expand --trace-format=json-v1 > build_trace.json
2. --trace-expand打印配置流程，分析冗余耗时流程
3. 排查find_package冗余调用，多次调用，在顶层统一调度
4. 排查存在其他的耗时操作，如file(GLOB_RECURSE ...)等
5. 对耗时点添加message打印时间戳进一步分析
6. 对复杂工程进行模块划分，独立模块使用external_project_add调度 + target粒度的大颗粒缓存
7. 如果是纯用户态编译单一工程，可以使用ninja作为编译器，配置流程会比使用make作为编译器效率高很多
8. variable_watch(MY_VAR)分析变量修改
运行态优化
1.使用ninja切换make
2.使用ccache等工具
3.预编译头文件target_precompile_headers
4.减少编译单元数量 Unity Build：set(CMAKE_UNITY_BUILD ON)
5. 减少容易依赖：减少不必要的 include_directories，降低编译器搜索开销。
6.-O编译选项的调整等
